  async function archiveSelectedAssignedPrograms(ids,studentId){
    if(!ids.length||!studentId||state.currentStudent?.id!==studentId)throw new Error('Student context changed. Reopen Edit Programs.');
    const allowed=new Set(activeStudentProgramsOrdered().filter(sp=>sp.student_id===studentId).map(sp=>sp.id));
    if(ids.some(id=>!allowed.has(id))||new Set(ids).size!==ids.length)throw new Error('Program selection changed. Reopen Edit Programs.');
    // Same soft removal as the existing one-at-a-time Remove control; never delete historic trials.
    const {data,error}=await sb.from('student_compass_programs')
      .update({status:'archived'})
      .eq('student_id',studentId)
      .in('id',ids)
      .neq('status','archived')
      .select('id');
    if(error)throw error;
    return Array.isArray(data)?data.length:0;
  }
  function openEditPrograms(){
    if(!state.currentStudent){toast('Open a student profile first.','error');return}
    const studentId=state.currentStudent.id;
    const rows=activeStudentProgramsOrdered().filter(sp=>sp.student_id===studentId);
    const selected=new Set();
    let processing=false;
    const body=rows.map((sp,i)=>{
      const p=programById(sp.program_id),name=p?.program_name||'Program',id=esc(sp.id);
      return '<div class="program-edit-row" data-program-edit-row="'+id+'">'+
        '<label class="program-edit-checkbox"><input type="checkbox" data-select-assigned="'+id+'" aria-label="Select '+esc(name)+'"></label>'+
        '<div class="program-edit-details"><strong>'+(i+1)+'. '+esc(name)+'</strong>'+
        '<small>'+esc(programDomain(p))+' · '+esc(roleName(sp.status))+' · Goal '+Number(sp.goal_percentage??80)+'%</small></div>'+
        '<div class="program-edit-actions">'+
        '<button class="btn move-btn" type="button" data-move-assigned="'+id+'" data-move-dir="-1" '+(i===0?'disabled':'')+' title="Move up">↑</button>'+
        '<button class="btn move-btn" type="button" data-move-assigned="'+id+'" data-move-dir="1" '+(i===rows.length-1?'disabled':'')+' title="Move down">↓</button>'+
        '<button class="btn danger" type="button" data-remove-assigned="'+id+'">Remove</button></div></div>';
    }).join('');
    const toolbar='<div class="program-bulk-toolbar" aria-label="Bulk program selection">'+
      '<button class="btn small" type="button" id="programSelectAll" '+(rows.length?'':'disabled')+'>Select All</button>'+
      '<button class="btn small" type="button" id="programClearSelection" disabled>Clear Selection</button>'+
      '<span id="programSelectionCount" aria-live="polite">0 selected</span>'+
      '<button class="btn danger" type="button" id="programBulkRemove" disabled>Remove Selected (0)</button></div>';
    openModal('Edit Programs',toolbar+
      '<div id="editProgramsList" class="program-edit-list">'+
      (body||'<div class="empty-state compact">No assigned programs.</div>')+'</div>'+
      '<div class="program-edit-note">Select multiple programs to remove them together. Removing an assignment keeps all prior session, trial, and graph history. Use ↑ and ↓ to change order.</div>',
      ()=>{
        const list=$('editProgramsList'),all=$('programSelectAll'),clear=$('programClearSelection');
        const counter=$('programSelectionCount'),bulk=$('programBulkRemove');
        if(!list||!all||!clear||!counter||!bulk)return;
        function refreshSelection(){
          list.querySelectorAll('input[data-select-assigned]').forEach(input=>{
            const checked=selected.has(input.dataset.selectAssigned);
            input.checked=checked;
            input.closest('.program-edit-row')?.classList.toggle('program-edit-row-selected',checked);
            input.disabled=processing;
          });
          counter.textContent=selected.size+' of '+rows.length+' selected';
          clear.disabled=processing||selected.size===0;
          all.disabled=processing||selected.size===rows.length;
          bulk.disabled=processing||selected.size===0;
          bulk.textContent=processing?'Removing…':'Remove Selected ('+selected.size+')';
          list.querySelectorAll('button').forEach(btn=>{btn.disabled=processing||btn.hasAttribute('data-original-disabled')});
        }
        // Move buttons retain their original disabled state after busy/idle transitions.
        list.querySelectorAll('button:disabled').forEach(btn=>btn.dataset.originalDisabled='true');
        list.addEventListener('change',e=>{
          const input=e.target.closest('input[data-select-assigned]');
          if(!input||processing)return;
          if(input.checked)selected.add(input.dataset.selectAssigned);
          else selected.delete(input.dataset.selectAssigned);
          refreshSelection();
        });
        all.addEventListener('click',()=>{
          if(processing)return;
          rows.forEach(sp=>selected.add(sp.id));
          refreshSelection();
        });
        clear.addEventListener('click',()=>{
          if(processing)return;
          selected.clear();refreshSelection();
        });
        bulk.addEventListener('click',async()=>{
          if(processing||!selected.size)return;
          const ids=[...selected],names=rows.filter(sp=>selected.has(sp.id))
            .map(sp=>programById(sp.program_id)?.program_name||'Program');
          const preview=names.slice(0,5).join(', ')+(names.length>5?' and '+(names.length-5)+' more':'');
          if(!confirm('Remove '+ids.length+' selected programs from this student?\n\n'+preview+
            '\n\nThese assignments will be archived. Past trial data, sessions, and graphs will be preserved.'))return;
          if(state.currentStudent?.id!==studentId){toast('The student profile changed. Please reopen Edit Programs.','error');return}
          processing=true;refreshSelection();
          try{
            const changed=await archiveSelectedAssignedPrograms(ids,studentId);
            closeModal();
            await loadStudentProfile();
            openEditPrograms();
            if(changed!==ids.length){
              toast(changed+' of '+ids.length+' selected programs were removed. Review the remaining assignments.','error');
            }else{
              toast(changed+' programs removed from the student. Past data was preserved.','success');
            }
          }catch(error){
            processing=false;refreshSelection();
            toast(error?.message||'Could not remove selected programs. Please try again.','error');
          }
        });
        list.addEventListener('click',async e=>{
          if(processing)return;
          const move=e.target.closest('[data-move-assigned]');
          if(move){await moveAssignedProgram(move.dataset.moveAssigned,Number(move.dataset.moveDir));return}
          const remove=e.target.closest('[data-remove-assigned]');
          if(remove){await removeAssignedProgram(remove.dataset.removeAssigned);return}
        });
        refreshSelection();
      });
  }
